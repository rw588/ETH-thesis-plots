% ====================================================================
% FIGURE 1 — one g-vector, cone/ring, single Gaussian constraint plane
% Coordinate mapping: (x,y,z) = (a1, a2, a0)  so a0 points up (z-axis)
% g = (1, cos theta, sin theta) in (a0,a1,a2) -> (cos theta, sin theta, 1) in (x,y,z)
% Ring traces a horizontal circle at z=a0=1
% ====================================================================
\tdplotsetmaincoords{65}{125}
\begin{tikzpicture}[tdplot_main_coords, scale=2.5,
    line cap=round, line join=round, font=\small]

  % g at theta=30 deg; |g|=sqrt(2)
  % In (x,y,z)=(a1,a2,a0): g = (cos theta, sin theta, 1)
  \def\gTheta{30}
  \pgfmathsetmacro{\GX}{cos(\gTheta)}
  \pgfmathsetmacro{\GY}{sin(\gTheta)}
  \pgfmathsetmacro{\GZ}{1.0}
  % unit vector ghat in (x,y,z)
  \pgfmathsetmacro{\GHX}{cos(\gTheta)/sqrt(2)}
  \pgfmathsetmacro{\GHY}{sin(\gTheta)/sqrt(2)}
  \pgfmathsetmacro{\GHZ}{1.0/sqrt(2)}
  % two orthonormal vectors spanning the plane (in new coords):
  %   e1 = (-sin theta, cos theta, 0)
  %   e2 = (-cos theta/sqrt2, -sin theta/sqrt2, 1/sqrt2)
  \pgfmathsetmacro{\EAX}{-sin(\gTheta)}
  \pgfmathsetmacro{\EAY}{cos(\gTheta)}
  \pgfmathsetmacro{\EAZ}{0.0}
  \pgfmathsetmacro{\EBX}{-cos(\gTheta)/sqrt(2)}
  \pgfmathsetmacro{\EBY}{-sin(\gTheta)/sqrt(2)}
  \pgfmathsetmacro{\EBZ}{1.0/sqrt(2)}
  % plane centre at t=0.75 along g (in new coords)
  \pgfmathsetmacro{\PCX}{0.75*cos(\gTheta)}
  \pgfmathsetmacro{\PCY}{0.75*sin(\gTheta)}
  \pgfmathsetmacro{\PCZ}{0.75}
  \def\hs{0.55}   % half-size of visible plane patch

  %-- Axes: x=a1, y=a2, z=a0 --
  \draw[->,thick,black!55] (0,0,0) -- (2.10,0,0) node[right=5pt]{$a_1$};
  \draw[->,thick,black!55] (0,0,0) -- (0,1.55,0) node[right=5pt]{$a_2$};
  \draw[->,thick,black!55] (0,0,0) -- (0,0,2.50) node[above=5pt]{$a_0$};

  %-- Cone lines origin -> ring --
  \foreach \cang in {0,30,...,330}{
    \draw[ConeRing!20,very thin] (0,0,0) -- ({2*cos(\cang)},{2*sin(\cang)},{2});
  }

  %-- Ring of possible g-vectors: 2*(cos t, sin t, 1) --
  \draw[ConeRing, thick, dashed,
        domain=0:360, smooth, samples=72, variable=\aa]
    plot ({2*cos(\aa)},{2*sin(\aa)},{2});

  %-- Gaussian constraint-plane layers back-to-front --
  %   sigma~0.10; opacities ~ exp(-(d/0.1)^2/2) / 7 (very low)
  %   For this view angle positive d is toward viewer
  \foreach \dVal/\fillop in {
      -0.24/0.006,
      -0.16/0.02,
      -0.08/0.05,
       0.00/0.10,
       0.08/0.05,
       0.16/0.02,
       0.24/0.006}{
    \pgfmathsetmacro{\OX}{\dVal*\GHX}
    \pgfmathsetmacro{\OY}{\dVal*\GHY}
    \pgfmathsetmacro{\OZ}{\dVal*\GHZ}
    \pgfmathsetmacro{\LAX}{\PCX + \hs*\EAX + \hs*\EBX + \OX}
    \pgfmathsetmacro{\LAY}{\PCY + \hs*\EAY + \hs*\EBY + \OY}
    \pgfmathsetmacro{\LAZ}{\PCZ + \hs*\EAZ + \hs*\EBZ + \OZ}
    \pgfmathsetmacro{\LBX}{\PCX - \hs*\EAX + \hs*\EBX + \OX}
    \pgfmathsetmacro{\LBY}{\PCY - \hs*\EAY + \hs*\EBY + \OY}
    \pgfmathsetmacro{\LBZ}{\PCZ - \hs*\EAZ + \hs*\EBZ + \OZ}
    \pgfmathsetmacro{\LCX}{\PCX - \hs*\EAX - \hs*\EBX + \OX}
    \pgfmathsetmacro{\LCY}{\PCY - \hs*\EAY - \hs*\EBY + \OY}
    \pgfmathsetmacro{\LCZ}{\PCZ - \hs*\EAZ - \hs*\EBZ + \OZ}
    \pgfmathsetmacro{\LDX}{\PCX + \hs*\EAX - \hs*\EBX + \OX}
    \pgfmathsetmacro{\LDY}{\PCY + \hs*\EAY - \hs*\EBY + \OY}
    \pgfmathsetmacro{\LDZ}{\PCZ + \hs*\EAZ - \hs*\EBZ + \OZ}
    \fill[GVecA, fill opacity=\fillop]
      (\LAX,\LAY,\LAZ) -- (\LBX,\LBY,\LBZ) --
      (\LCX,\LCY,\LCZ) -- (\LDX,\LDY,\LDZ) -- cycle;
  }

  %-- Border of the d=0 central plane --
  \pgfmathsetmacro{\BAX}{\PCX + \hs*\EAX + \hs*\EBX}
  \pgfmathsetmacro{\BAY}{\PCY + \hs*\EAY + \hs*\EBY}
  \pgfmathsetmacro{\BAZ}{\PCZ + \hs*\EAZ + \hs*\EBZ}
  \pgfmathsetmacro{\BBX}{\PCX - \hs*\EAX + \hs*\EBX}
  \pgfmathsetmacro{\BBY}{\PCY - \hs*\EAY + \hs*\EBY}
  \pgfmathsetmacro{\BBZ}{\PCZ - \hs*\EAZ + \hs*\EBZ}
  \pgfmathsetmacro{\BCX}{\PCX - \hs*\EAX - \hs*\EBX}
  \pgfmathsetmacro{\BCY}{\PCY - \hs*\EAY - \hs*\EBY}
  \pgfmathsetmacro{\BCZ}{\PCZ - \hs*\EAZ - \hs*\EBZ}
  \pgfmathsetmacro{\BDX}{\PCX + \hs*\EAX - \hs*\EBX}
  \pgfmathsetmacro{\BDY}{\PCY + \hs*\EAY - \hs*\EBY}
  \pgfmathsetmacro{\BDZ}{\PCZ + \hs*\EAZ - \hs*\EBZ}
  \draw[GVecA!70!black, semithick]
    (\BAX,\BAY,\BAZ) -- (\BBX,\BBY,\BBZ) --
    (\BCX,\BCY,\BCZ) -- (\BDX,\BDY,\BDZ) -- cycle;

  %-- Dashed normal axis through the plane centre --
  \draw[black!35, thin, dashed]
    ({\PCX-0.30*\GHX},{\PCY-0.30*\GHY},{\PCZ-0.30*\GHZ}) --
    ({\PCX+0.30*\GHX},{\PCY+0.30*\GHY},{\PCZ+0.30*\GHZ});

  %-- g-vector arrow --
  \draw[-{Stealth[length=7pt,width=5pt]},very thick,GVecA]
    (0,0,0) -- ({2*\GX},{2*\GY},{2*\GZ})
    node[above=5pt,font=\normalsize]{$\vec{g}_1$};

  %-- Dot where the g-vector pierces its own plane (at t=0.75 along g) --
  \fill[GVecA] (\PCX,\PCY,\PCZ) circle (2pt);

  %-- C-optimality vector v=(0,a2,-a1) evaluated at the g-tip --
  %   (a0,a1,a2)=(1,cos theta,sin theta) at tip
  %   v=(0,sin theta,-cos theta) in (a0,a1,a2) = (sin theta,-cos theta,0) in (x,y,z)
  %   tangent to the g-vector ring at the tip
  \pgfmathsetmacro{\VX}{sin(\gTheta)}
  \pgfmathsetmacro{\VY}{-cos(\gTheta)}
  \draw[-{Stealth[length=6pt,width=4pt]},thick,CVecCol]
    ({2*\GX},{2*\GY},{2*\GZ}) --
    ({2*\GX+0.55*\VX},{2*\GY+0.55*\VY},{2*\GZ})
    node[above=5pt]{$\vec{v}$};

  \fill[black] (0,0,0) circle (1pt);

\end{tikzpicture}

% ====================================================================
% FIGURE 2 — three g-vectors (0, 120, 240 deg), three Gaussian planes,
%            and the Cramer-Rao error ellipsoid at their intersection.
%
% Fisher info: F = sum_i g_i g_i^T = diag(3, 3/2, 3/2)  (3 equal weights)
% Covariance:  Sigma = F^{-1} = diag(1/3, 2/3, 2/3)
% Ellipsoid:   3 a0^2 + 1.5 a1^2 + 1.5 a2^2 = 1
%              semi-axes: r0 = 1/sqrt(3) ~ 0.577 (along a0),
%                         r12 = sqrt(2/3) ~ 0.816 (along a1, a2)
% Coordinate mapping: (x,y,z) = (a1, a2, a0)
% ====================================================================
\tdplotsetmaincoords{65}{125}
\begin{tikzpicture}[tdplot_main_coords, scale=2.1,
    line cap=round, line join=round, font=\small]

  \def\phs{0.96}    % half-size of each plane patch
  \def\ptval{0.55}  % planes centred at ptval * g

  %-- Axes: x=a1, y=a2, z=a0 --
  \draw[->,thick,black!55] (0,0,0) -- (2.30,0,0) node[right=5pt]{$a_1$};
  \draw[->,thick,black!55] (0,0,0) -- (0,1.95,0) node[right=5pt]{$a_2$};
  \draw[->,thick,black!55] (0,0,0) -- (0,0,2.50) node[above=5pt]{$a_0$};

  %-- Ring joining the tips of the g-vectors at z=2 --
  \draw[ConeRing, thick, dashed,
        domain=0:360, smooth, samples=72, variable=\aa]
    plot ({2*cos(\aa)},{2*sin(\aa)},{2});

  % Intersection of the 3 planes: a0=2t, a1=a2=0 => (x,y,z)=(0,0,2t)
  % Centre each plane patch there so the crossing point is always visible
  \pgfmathsetmacro{\ez}{2.0*\ptval}

  %-- Gaussian planes: depth order for this view: theta=0 farthest, then 240, then 120 --
  %   plane colour = g-vector colour for each plane
  \foreach \gAngle/\pcol in {0/GVecA, 240/GVecC, 120/GVecB}{
    % g = (cos theta, sin theta, 1) in (x,y,z)
    \pgfmathsetmacro{\GHX}{cos(\gAngle)/sqrt(2)}
    \pgfmathsetmacro{\GHY}{sin(\gAngle)/sqrt(2)}
    \pgfmathsetmacro{\GHZ}{1.0/sqrt(2)}
    % e1 = (-sin theta, cos theta, 0)
    % e2 = (-cos theta/sqrt2, -sin theta/sqrt2, 1/sqrt2)
    \pgfmathsetmacro{\EAX}{-sin(\gAngle)}
    \pgfmathsetmacro{\EAY}{cos(\gAngle)}
    \pgfmathsetmacro{\EAZ}{0.0}
    \pgfmathsetmacro{\EBX}{-cos(\gAngle)/sqrt(2)}
    \pgfmathsetmacro{\EBY}{-sin(\gAngle)/sqrt(2)}
    \pgfmathsetmacro{\EBZ}{1.0/sqrt(2)}
    % centre patch at the intersection point so the crossing is always inside
    \pgfmathsetmacro{\PCX}{0.0}
    \pgfmathsetmacro{\PCY}{0.0}
    \pgfmathsetmacro{\PCZ}{\ez}

    \foreach \dVal/\fillop in {
        -0.22/0.001, -0.14/0.006, -0.07/0.012,
         0.00/0.020,
         0.07/0.012,  0.14/0.006,  0.22/0.001}{
      \pgfmathsetmacro{\OX}{\dVal*\GHX}
      \pgfmathsetmacro{\OY}{\dVal*\GHY}
      \pgfmathsetmacro{\OZ}{\dVal*\GHZ}
      \pgfmathsetmacro{\LAX}{\PCX + \phs*\EAX + \phs*\EBX + \OX}
      \pgfmathsetmacro{\LAY}{\PCY + \phs*\EAY + \phs*\EBY + \OY}
      \pgfmathsetmacro{\LAZ}{\PCZ + \phs*\EAZ + \phs*\EBZ + \OZ}
      \pgfmathsetmacro{\LBX}{\PCX - \phs*\EAX + \phs*\EBX + \OX}
      \pgfmathsetmacro{\LBY}{\PCY - \phs*\EAY + \phs*\EBY + \OY}
      \pgfmathsetmacro{\LBZ}{\PCZ - \phs*\EAZ + \phs*\EBZ + \OZ}
      \pgfmathsetmacro{\LCX}{\PCX - \phs*\EAX - \phs*\EBX + \OX}
      \pgfmathsetmacro{\LCY}{\PCY - \phs*\EAY - \phs*\EBY + \OY}
      \pgfmathsetmacro{\LCZ}{\PCZ - \phs*\EAZ - \phs*\EBZ + \OZ}
      \pgfmathsetmacro{\LDX}{\PCX + \phs*\EAX - \phs*\EBX + \OX}
      \pgfmathsetmacro{\LDY}{\PCY + \phs*\EAY - \phs*\EBY + \OY}
      \pgfmathsetmacro{\LDZ}{\PCZ + \phs*\EAZ - \phs*\EBZ + \OZ}
      \fill[\pcol, fill opacity=\fillop]
        (\LAX,\LAY,\LAZ) -- (\LBX,\LBY,\LBZ) --
        (\LCX,\LCY,\LCZ) -- (\LDX,\LDY,\LDZ) -- cycle;
    }

    % border of the d=0 plane
    \pgfmathsetmacro{\BAX}{\PCX + \phs*\EAX + \phs*\EBX}
    \pgfmathsetmacro{\BAY}{\PCY + \phs*\EAY + \phs*\EBY}
    \pgfmathsetmacro{\BAZ}{\PCZ + \phs*\EAZ + \phs*\EBZ}
    \pgfmathsetmacro{\BBX}{\PCX - \phs*\EAX + \phs*\EBX}
    \pgfmathsetmacro{\BBY}{\PCY - \phs*\EAY + \phs*\EBY}
    \pgfmathsetmacro{\BBZ}{\PCZ - \phs*\EAZ + \phs*\EBZ}
    \pgfmathsetmacro{\BCX}{\PCX - \phs*\EAX - \phs*\EBX}
    \pgfmathsetmacro{\BCY}{\PCY - \phs*\EAY - \phs*\EBY}
    \pgfmathsetmacro{\BCZ}{\PCZ - \phs*\EAZ - \phs*\EBZ}
    \pgfmathsetmacro{\BDX}{\PCX + \phs*\EAX - \phs*\EBX}
    \pgfmathsetmacro{\BDY}{\PCY + \phs*\EAY - \phs*\EBY}
    \pgfmathsetmacro{\BDZ}{\PCZ + \phs*\EAZ - \phs*\EBZ}
    \draw[\pcol!70!black, semithick]
      (\BAX,\BAY,\BAZ) -- (\BBX,\BBY,\BBZ) --
      (\BCX,\BCY,\BCZ) -- (\BDX,\BDY,\BDZ) -- cycle;
  }

  %-- Cramer-Rao error ellipsoid drawn first so g arrows render on top --
  %   rZ = 1/sqrt(3) ~ 0.577 along z=a0 (shorter axis)
  %   rOne = sqrt(2/3) ~ 0.816 along x=a1, y=a2 (longer axes)
  \pgfmathsetmacro{\rZ}{1.0/sqrt(3)}
  \pgfmathsetmacro{\rOne}{sqrt(2.0/3.0)}

  % Shaded ellipsoid: stacked filled latitude discs (back to front) + equatorial outline
  \foreach \phiDeg/\fillop in {
      -80/0.03, -65/0.05, -50/0.07, -35/0.09,
      -20/0.11,  -5/0.13,  10/0.14,  25/0.13,
       40/0.11,  55/0.09,  70/0.06,  85/0.03}{
    \pgfmathsetmacro{\latR}{\rOne*cos(\phiDeg)}
    \pgfmathsetmacro{\latZ}{\ez+\rZ*sin(\phiDeg)}
    \draw[EllipCol, draw opacity=0, fill=EllipCol, fill opacity=\fillop,
          domain=0:360, smooth, samples=36, variable=\aa]
      plot ({\latR*cos(\aa)},{\latR*sin(\aa)},{\latZ});
  }
  \draw[EllipCol!80!black, semithick,
        domain=0:360, smooth, samples=60, variable=\aa]
    plot ({\rOne*cos(\aa)},{\rOne*sin(\aa)},{\ez});

  % dot at intersection / ellipsoid centre
  \fill[EllipCol!80!black] (0,0,\ez) circle (1.5pt);

  %-- g-vector arrows and crossing dots --
  \foreach \gAngle/\gcol/\glab/\glabpos in {
      0/GVecA/{\vec{g}_1}/{above},
    120/GVecB/{\vec{g}_2}/{above left},
    240/GVecC/{\vec{g}_3}/{below left}}{
    \pgfmathsetmacro{\GTIPX}{cos(\gAngle)}
    \pgfmathsetmacro{\GTIPY}{sin(\gAngle)}
    \draw[-{Stealth[length=6pt,width=4pt]},thick,\gcol]
      (0,0,0) -- ({2*\GTIPX},{2*\GTIPY},{2})
      node[\glabpos]{$\glab$};
    % dot where this g-vector pierces its own plane
    \pgfmathsetmacro{\dotx}{\ptval*cos(\gAngle)}
    \pgfmathsetmacro{\doty}{\ptval*sin(\gAngle)}
    \fill[\gcol] (\dotx,\doty,\ptval) circle (2pt);
  }

  %-- Single C-optimality vector from ellipsoid centre --
  %   same direction as Fig 1: v=(0,sin30,-cos30) in (a0,a1,a2)
  %   = (sin30,-cos30,0) in (x,y,z) = (0.5,-0.866,0)
  \draw[-{Stealth[length=6pt,width=4pt]},thick,CVecCol]
    (0,0,\ez) -- ({1.35*sin(30)},{-1.35*cos(30)},\ez)
    node[left=5pt]{$\vec{v}$};

  \fill[black] (0,0,0) circle (1pt);

\end{tikzpicture}
